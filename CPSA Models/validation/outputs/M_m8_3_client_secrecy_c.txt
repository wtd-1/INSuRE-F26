        (enc (cat c cpk (enc (hash c cpk) (privk ca)))

            (enc (hash c cpk) (privk ca)))
          (hash (exp (gen) (mul x y)) "c hs traffic" cr sr)))

        (enc request
          (hash "c ap traffic" cr sr
            (hash (exp (gen) (mul x y)) "derived"))))

        (enc response
          (hash "s ap traffic" cr sr
            (hash (exp (gen) (mul x y)) "derived")))))

    (forall ((z0 z1 strd) (c name))
